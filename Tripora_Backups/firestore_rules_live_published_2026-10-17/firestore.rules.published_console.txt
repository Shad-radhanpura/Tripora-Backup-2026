// =====================================================================
// PROPOSED Firestore security rules for Tripora (project tripora-7ded0)
// ---------------------------------------------------------------------
// Revised 15 Oct 2026 (permission-denied diagnosis). NOT deployed and NOT
// copied into firestore.rules. Backup of the previous version:
//   _backup_rules_fix_2026-10-15\firestore.rules.proposed.orig
//
// To use (only after you approve):
//   1. Test in Firebase Console > Firestore > Rules > Rules Playground
//      (checklist in the chat / README), or with the emulator test in
//      firestore_rules_test\.
//   2. Copy this file's contents into firestore.rules.
//   3. firebase deploy --only firestore:rules --project tripora-7ded0
//
// Admin accounts are made by hand in the Firebase Console:
//   Firestore > collection "admins" > Document ID = the account's UID
//   (Authentication > Users > "User UID"), field role (string) = admin
// The Console is not limited by these rules; the app can never write here.
// =====================================================================

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isOwner(uid) {
      return signedIn() && request.auth.uid == uid;
    }

    function adminDoc() {
      return /databases/$(database)/documents/admins/$(request.auth.uid);
    }

    // Admin = an admins/{uid} document whose role is exactly "admin".
    // Same test as FirestoreService.isAdmin() in the app.
    function isAdmin() {
      return signedIn()
        && exists(adminDoc())
        && get(adminDoc()).data.get('role', '') == 'admin';
    }

    // ---------------- admins/{uid} ----------------
    // THE ROLE CHECK. A signed-in user may read only the document whose ID
    // equals their own UID (isOwner), which is exactly what
    // FirestoreService.isAdmin(uid) reads after login. Admins may also list
    // the collection (Admin Panel "admins" count). Nobody can create, edit or
    // delete an admin document from the app, so no one can promote
    // themselves; a missing document simply means "not an admin".
    match /admins/{uid} {
      allow get: if isOwner(uid) || isAdmin();
      allow list: if isAdmin();
      allow create, update, delete: if false;
    }

    // ---------------- users/{uid} ----------------
    // Signup writes: uid, name, email, createdAt, lastLoginAt, loginCount.
    // Each login (merge) writes: uid, email, name, lastLoginAt, loginCount.
    match /users/{uid} {
      allow read: if isOwner(uid) || isAdmin();

      allow create: if isOwner(uid)
        && request.resource.data.keys().hasOnly(
             ['uid', 'name', 'email', 'createdAt', 'lastLoginAt', 'loginCount',
              'phone', 'homeCity'])
        && request.resource.data.uid == uid
        && (!('createdAt' in request.resource.data)
            || request.resource.data.createdAt == request.time);

      // Own profile only; createdAt can never change and no extra fields
      // (such as "role" or "isAdmin") can be added.
      allow update: if isOwner(uid)
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
             ['uid', 'name', 'email', 'lastLoginAt', 'loginCount',
              'phone', 'homeCity'])
        && request.resource.data.uid == uid;

      allow delete: if false;

      // ---------- users/{uid}/favourites/{destinationId} ----------
      match /favourites/{destinationId} {
        allow read, delete: if isOwner(uid);
        allow create, update: if isOwner(uid)
          && request.resource.data.destinationId == destinationId
          && request.resource.data.keys().hasOnly(
               ['destinationId', 'name', 'location', 'imageUrl', 'addedAt']);
      }
    }

    // Admin Panel: count favourites across all users (collection group).
    match /{path=**}/favourites/{destinationId} {
      allow read: if isAdmin();
    }

    // ---------------- trips/{tripId} ----------------
    match /trips/{tripId} {
      // Works for single reads and for the app's
      // where('userId', isEqualTo: uid) queries; admins can read all.
      allow read: if signedIn()
        && (resource.data.userId == request.auth.uid || isAdmin());

      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(
             ['userId', 'destination', 'budget', 'days', 'interest',
              'itinerary', 'createdAt'])
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.destination is string
        && request.resource.data.itinerary is string
        && request.resource.data.days is int;

      // The app never edits trips. Owners (or admins) may delete.
      allow update: if false;
      allow delete: if signedIn()
        && (resource.data.userId == request.auth.uid || isAdmin());
    }

    // ---------------- packages/{packageId} ----------------
    // Tour packages managed in Admin Panel > Package Management. Every
    // signed-in user can read them; only admins can write.
    match /packages/{packageId} {
      allow read: if signedIn();
      allow create, update: if isAdmin()
        && request.resource.data.keys().hasOnly(
             ['title', 'durationDays', 'cities', 'region', 'category',
              'pricePerPerson', 'hotelClass', 'facilities', 'highlights',
              'itinerary', 'description', 'imageUrl', 'active'])
        && request.resource.data.title is string
        && request.resource.data.durationDays in [10, 15, 20, 30]
        && request.resource.data.pricePerPerson is int
        && request.resource.data.active is bool;
      allow delete: if isAdmin();
    }

    // ---------------- bookings/{bookingId} ----------------
    // DEMO booking requests. A user creates their own request (always
    // "pending" and isDemo == true) and may cancel it; only admins can
    // confirm, complete, cancel or delete any booking.
    match /bookings/{bookingId} {
      allow read: if signedIn()
        && (resource.data.userId == request.auth.uid || isAdmin());

      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(
             ['userId', 'userEmail', 'kind', 'itemId', 'title', 'subtitle',
              'travelDate', 'travellers', 'unitPrice', 'totalPrice',
              'contactName', 'contactPhone', 'notes', 'details', 'status',
              'reference', 'isDemo', 'createdAt'])
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.status == 'pending'
        && request.resource.data.isDemo == true
        && request.resource.data.kind in
             ['package', 'hotel', 'flight', 'train', 'bus', 'cab']
        && request.resource.data.travellers is int
        && request.resource.data.travellers >= 1
        && request.resource.data.travellers <= 9
        && request.resource.data.unitPrice is int
        && request.resource.data.totalPrice ==
             request.resource.data.unitPrice * request.resource.data.travellers
        && request.resource.data.title is string
        && request.resource.data.notes.size() <= 300
        && request.resource.data.createdAt == request.time;

      allow update: if signedIn() && (
        // Owner: cancel an open request, nothing else.
        (resource.data.userId == request.auth.uid
          && request.resource.data.diff(resource.data).affectedKeys()
               .hasOnly(['status', 'updatedAt'])
          && request.resource.data.status == 'cancelled'
          && resource.data.status in ['pending', 'confirmed'])
        ||
        // Admin: move through the workflow and leave a note.
        (isAdmin()
          && request.resource.data.diff(resource.data).affectedKeys()
               .hasOnly(['status', 'updatedAt', 'adminNote'])
          && request.resource.data.status in
               ['pending', 'confirmed', 'completed', 'cancelled'])
      );

      allow delete: if isAdmin();
    }

    // Everything else is closed (no match = denied).
  }
}
